51// Fair random whole number from min to max, both included.
52function randomInt(min, max) {
53  return Math.floor(Math.random() * (max - min + 1)) + min;
54}
55
56function makeItem() {
57  return {
58    name: ITEM_NAMES[randomInt(0, ITEM_NAMES.length - 1)],
59    value: randomInt(MIN_PRICE, MAX_PRICE)
60  };
61}
62
63function loadBest() {
64  try {
65    const saved = Number(localStorage.getItem(BEST_KEY));
66    return Number.isInteger(saved) && saved >= 0 ? saved : 0;
67  } catch (e) {
68    return 0;
69  }
70}
71
72function saveBest(value) {
73  try {
74    localStorage.setItem(BEST_KEY, String(value));
75  } catch (e) {
76    /* storage can be blocked, the game still works */
77  }
78}
79
80function setMessage(text, kind) {
81  messageEl.textContent = text;
82  messageEl.className = 'message message--' + kind;
83}

6. Reading the move6. Разбор хода6. Жүрісті талдау linesстрокижолдар 101–111

101// ---------- a move ----------
102function normalizeGuess(raw) {
103  const text = String(raw).trim().toLowerCase();
104  if (text === 'h' || text === 'higher') {
105    return 'higher';
106  }
107  if (text === 'l' || text === 'lower') {
108    return 'lower';
109  }
110  return null;
111}
  • normalizeGuess turns h / higher / l / lower into a clean choice, anything else into null.normalizeGuess превращает h / higher / l / lower в чистый выбор, всё остальное в null.normalizeGuess h / higher / l / lower мәндерін таза таңдауға, қалғанын null-ға айналдырады.

7. makeGuess: the main function7. makeGuess: главная функция7. makeGuess: басты функция linesстрокижолдар 113–171

113function makeGuess(raw) {
114  if (state.over) {
115    setMessage('The game is over. Press "Restart game" to play again.', 'warn');
116    return;
117  }
118
119  const choice = normalizeGuess(raw);
120  if (choice === null) {
121    // invalid input: reject, state is not changed
122    typedInput.classList.add('typed-input--invalid');
123    setMessage('Invalid move. Type "higher" or "lower" (or H / L).', 'bad');
124    return;
125  }
126  typedInput.classList.remove('typed-input--invalid');
127
128  const previous = state.current;
129  const next = makeItem();
130  state.next = next;
131
132  // Equal values count as a right guess.
133  const correct =
134    (choice === 'higher' && next.value >= previous.value) ||
135    (choice === 'lower' && next.value <= previous.value);
136
137  let text;
138  if (correct) {
139    state.streak += 1;
140    if (state.streak > state.best) {
141      state.best = state.streak;
142      saveBest(state.best);
143    }
144    text = next.value === previous.value
145      ? 'Equal prices ($' + next.value + ') - counts as a right guess!'
146      : 'Right! $' + next.value + ' is ' + (next.value > previous.value ? 'higher' : 'lower') + ' than $' + previous.value + '.';
147    setMessage(text, 'ok');
148  } else {
149    state.lives -= 1;
150    state.streak = 0;
151    text = 'Wrong. $' + next.value + ' is ' + (next.value > previous.value ? 'higher' : 'lower') + ' than $' + previous.value + '.';
152    setMessage(text, 'bad');
153  }
154
155  // game end conditions
156  if (state.lives === 0) {
157    state.over = true;
158    state.won = false;
159  } else if (state.streak === WIN_STREAK) {
160    state.over = true;
161    state.won = true;
162  }
163
164  state.log.unshift({ text: text, correct: correct });
165  if (state.log.length > LOG_LIMIT) {
166    state.log.length = LOG_LIMIT;
167  }
168
169  state.current = next;
170  render();
171}
  • First the checks: game over or invalid input → message and returnСначала проверки: игра окончена или неверный ввод → сообщение и returnАлдымен тексерулер: ойын бітті немесе қате енгізу → хабарлама және return
  • correct is decided with ===, &&, ||; >= makes equal prices right.correct решается через ===, &&, ||; >= делает равные цены верными.correct ===, &&, || арқылы шешіледі; >= тең бағаны дұрыс етеді.
  • Right: streak + 1, update best. Wrong: lives − 1, streak 0.Верно: серия + 1, обновить рекорд. Неверно: жизни − 1, серия 0.Дұрыс: серия + 1, рекордты жаңарту. Қате: өмір − 1, серия 0.
  • End: lives === 0 loss, streak === WIN_STREAK win. Then log (max 5) and render().Конец: lives === 0 поражение, streak === WIN_STREAK победа. Затем лог (макс 5) и render().Соңы: lives === 0 жеңіліс, streak === WIN_STREAK жеңіс. Сосын лог (макс 5) және render().
