4. Validation rules4. Правила проверки4. Тексеру ережелері linesстрокижолдар 123–201

123// ---------- validation rules ----------
124// every rule returns an error text or '' when the value is correct
125const validators = {
126  event: function (value) {
127    return findEvent(value) ? '' : 'Choose an event from the list.';
128  },
129
130  date: function (value) {
131    if (value === '') {
132      return 'Choose a date.';
133    }
134    if (value < todayString()) {
135      return 'The date is in the past. Choose today or a later day.';
136    }
137    if (value > maxDateString()) {
138      return 'Tickets are sold only ' + MAX_DAYS_AHEAD + ' days ahead.';
139    }
140    if (SOLD_OUT_DATES.includes(value)) {
141      return 'Tickets for this date are sold out. Choose another date.';
142    }
143    return '';
144  },
145
146  type: function (value) {
147    return findType(value) ? '' : 'Choose a ticket type.';
148  },
149
150  quantity: function (value) {
151    if (value.trim() === '') {
152      return 'Enter how many tickets you need.';
153    }
154    const n = Number(value);
155    if (!Number.isInteger(n)) {
156      return 'Quantity must be a whole number.';
157    }
158    if (n < 1 || n > MAX_QUANTITY) {
159      return 'Quantity must be from 1 to ' + MAX_QUANTITY + '.';
160    }
161    const type = findType(controls.type.value);
162    if (type && n > type.seats) {
163      return 'Only ' + type.seats + ' seat(s) left for the ' + type.name + ' ticket.';
164    }
165    return '';
166  },
167
168  promo: function (value) {
169    if (value.trim() === '') {
170      return ''; // the promo code is optional
171    }
172    return findPromo(value) ? '' : 'Unknown promo code. Check it or leave the field empty.';
173  },
174
175  name: function (value) {
176    const text = value.trim();
177    if (text === '') {
178      return 'Enter your name and surname.';
179    }
180    // at least two words, letters only (Latin or Cyrillic), each word 2+ letters
181    if (!/^[\p{L}]{2,}(?:[ '-][\p{L}]{2,})+$/u.test(text)) {
182      return 'Use two or more words with letters only, e.g. "Aidos Nurlan".';
183    }
184    return '';
185  },
186
187  email: function (value) {
188    const text = value.trim();
189    if (text === '') {
190      return 'Enter your email.';
191    }
192    if (controls.email.validity.typeMismatch || !/^[^@\s]+@[^@\s]+\.[^@\s]{2,}$/.test(text)) {
193      return 'Email must look like name@example.com.';
194    }
195    const domain = text.split('@')[1].toLowerCase();
196    if (BLOCKED_EMAIL_DOMAINS.includes(domain)) {
197      return 'Temporary email services (' + domain + ') are not allowed.';
198    }
199    return '';
200  }
201};
  • Each rule is a function: it returns an error text, or an empty string if the value is correct.Каждое правило функция: возвращает текст ошибки или пустую строку, если значение верное.Әр ереже функция: қате мәтінін немесе мән дұрыс болса бос жолды қайтарады.
  • date: past, too far ahead, sold out. quantity: whole number, 1–10, not above seats left.date: прошлое, слишком далеко, sold out. quantity: целое, 1–10, не больше оставшихся мест.date: өткен, тым алыс, sold out. quantity: бүтін, 1–10, қалған орыннан аспайды.
  • name: regex with \p{L}, two or more words. email: format and blocked domains.name: регулярка с \p{L}, два и более слова. email: формат и запрещённые домены.name: \p{L} бар регулярка, екі және одан көп сөз. email: формат және тыйым салынған домендер.

242function calculateTotal(typeId, quantityText, promoText) {
243  const type = findType(typeId);
244  const qty = Number(quantityText);
245  if (!type || !Number.isInteger(qty) || qty < 1) {
246    return { total: 0, discount: 0 };
247  }
248  const promo = findPromo(promoText);
249  const discount = promo ? promo.discount : 0;
250  const gross = type.price * qty;
251  return { total: Math.round(gross * (100 - discount) / 100), discount: discount };
252}

288// ---------- events ----------
289for (const fieldName of FIELD_NAMES) {
290  const control = controls[fieldName];
291
292  // 1) leaving a field: first check
293  control.addEventListener('blur', function () {
294    touched.add(fieldName);
295    checkField(fieldName);
296  });
297
298  // 2) after that: check on every input
299  control.addEventListener('input', function () {
300    if (touched.has(fieldName)) {
301      checkField(fieldName);
302    }
303    // quantity depends on the ticket type, so check it again when the type changes
304    if (fieldName === 'type' && touched.has('quantity')) {
305      checkField('quantity');
306    }
307    renderPreview();
308  });
309}
310
311form.addEventListener('submit', function (event) {
312  event.preventDefault();
313
314  let errorCount = 0;
315  let firstInvalid = null;
316
317  for (const fieldName of FIELD_NAMES) {
318    touched.add(fieldName);
319    if (checkField(fieldName) !== '') {
320      errorCount += 1;
321      if (firstInvalid === null) {
322        firstInvalid = controls[fieldName];
323      }
324    }
325  }
326
327  if (errorCount > 0) {
328    summaryEl.textContent = 'The form has ' + errorCount + ' error' + (errorCount === 1 ? '' : 's') + '. Please fix the fields marked in red.';
329    summaryEl.classList.remove('hidden');
330    firstInvalid.focus();
331    return;
332  }
333
334  summaryEl.classList.add('hidden');
335  summaryEl.textContent = '';
336  buildFinalTicket();
337});
338
339form.addEventListener('reset', function () {
340  // wait until the browser clears the values, then clean the states
341  setTimeout(function () {
342    resetFormState();
343  }, 0);
344});

  • input: check only if touched; re-check quantity when type changes; always update the preview.input: проверять только touched; при смене type перепроверить quantity; предпросмотр обновлять всегда.input: тек touched болса тексеру; type өзгерсе quantity-ді қайта тексеру; көріністі әрдайым жаңарту.

347function makeBookingCode() {
348  let code = 'AK-';
349  for (let i = 0; i < CODE_LENGTH; i++) {
350    code += CODE_ALPHABET[randomInt(0, CODE_ALPHABET.length - 1)];
351  }
352  return code;
353}

355function buildFinalTicket() {
356  const ev = findEvent(controls.event.value);
357  const type = findType(controls.type.value);
358  const qty = Number(controls.quantity.value);
359  const money = calculateTotal(controls.type.value, controls.quantity.value, controls.promo.value);
360  const data = {
361    event: ev.name,
362    date: controls.date.value,
363    type: type.name,
364    qty: qty,
365    name: controls.name.value.trim(),
366    email: controls.email.value.trim(),
367    discount: money.discount,
368    total: money.total
369  };
370
371  type.seats -= qty; // seats left go down after a booking
372
373  resultEl.textContent = '';
374  const ticket = document.createElement('div');
375  ticket.className = 'ticket ticket--final';

378  heading.className = 'ticket__title';

380  ticket.appendChild(heading);

382  const rows = document.createElement('dl');
383  rows.className = 'ticket__rows';
384  addRow(rows, 'Date', data.date);
385  addRow(rows, 'Type', data.type);
386  addRow(rows, 'Quantity', String(data.qty));
387  addRow(rows, 'Name', data.name);
388  addRow(rows, 'Email', data.email);
