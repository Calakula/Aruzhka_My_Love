52function countInColumn(key) {

62// my own loop: total hours of cards that are not finished yet
63function remainingHours() {
64  let total = 0;
65  for (let i = 0; i < tasks.length; i++) {
66    if (tasks[i].column !== 'done') {
67      total += tasks[i].hours;
68    }
69  }
70  return total;

73function columnIndex(key) {
74  for (let i = 0; i < COLUMNS.length; i++) {
75    if (COLUMNS[i].key === key) {
76      return i;
77    }
78  }
79  return -1;

82function matchesFilters(task) {
83  const text = query.trim().toLowerCase();
84  const titleOk = text === '' || task.title.toLowerCase().includes(text);
85  const tagOk = tagFilter === ALL_TAGS || task.tag === tagFilter;
86  return titleOk && tagOk;
87}

89function compareTasks(a, b) {
90  if (sortBy === 'hours') {

93  if (sortBy === 'title') {

95  }
96  return a.priority - b.priority || a.title.localeCompare(b.title);
97}

4. moveTask: the only data change4. moveTask: единственное изменение данных4. moveTask: деректің жалғыз өзгеруі linesстрокижолдар 103–127

103// ---------- the ONLY place that changes the data: move ----------
104function moveTask(id, direction) {
105  const task = tasks.find(function (t) { return t.id === id; });
106  if (!task) {
107    return;
108  }
109  const from = columnIndex(task.column);
110  const to = from + direction;
111
112  if (to < 0 || to >= COLUMNS.length) {
113    setMessage('"' + task.title + '" cannot go further in that direction.', 'warn');
114    return;
115  }
116
117  const target = COLUMNS[to].key;
118
119  // rule: the middle column holds at most DOING_LIMIT cards
120  if (target === 'doing' && countInColumn('doing') >= DOING_LIMIT) {
121    setMessage('Blocked: "In progress" already has ' + DOING_LIMIT + ' cards. Finish one of them first.', 'bad');
122    return;
123  }
124
125  task.column = target;
126  setMessage('Moved "' + task.title + '" to "' + COLUMNS[to].title + '".', 'ok');
127}
  • find gets the card; to = from + direction is the next column.find находит карточку; to = from + direction следующая колонка.find карточканы табады; to = from + direction келесі баған.
  • Rule: if the target is “doing” and it already has the limit → message and return.Правило: если цель «doing» и там уже лимит → сообщение и return.Ереже: мақсат «doing» және лимит толса → хабарлама және return.
  • Otherwise change task.column.Иначе меняем task.column.Әйтпесе task.column өзгереді.

256function fillTagOptions() {
257  const tags = [];

271    option.value = tag;

275}
